** Column limit
*** Display column limit
120 is the default column limit, can change it with ~C-x f~
#+begin_src emacs-lisp
  (setopt whitespace-line-column 120)
  (add-hook 'prog-mode-hook #'display-fill-column-indicator-mode)
  (setopt fill-column 120)
#+end_src
*** Function to change the column limit
#+begin_src emacs-lisp
  (defun change-column-limit (new-limit)
    "Change the column limit and reload whitespace mode"
    (interactive "nEnter new column limit value: ")
    (setopt fill-column new-limit)
    (setopt whitespace-line-column new-limit)
    (whitespace-mode 0)
    (whitespace-mode 1))

  (global-set-key (kbd "C-x f") 'change-column-limit)
#+end_src
